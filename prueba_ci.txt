start PruebaCodIntermedio {
    // ==============================
    // 1. Declaraciones de variables
    // ==============================
    @ a int;
    @ b int;
    @ c int;
    @ d int;
    @ resultado int;
    @ contador int;

    // ==============================
    // 2. Asignaciones simples (:=)
    // ==============================
    a = 1;
    b = 2;
    c = 3;
    d = 4;

    // ==============================
    // 3. Expresiones aritmeticas (cuadruplos *, +, -)
    // ==============================
    // Esperado:
    //   (*, b, c, T1)
    //   (+, a, T1, T2)
    //   (-, T2, d, T3)
    //   (:=, T3, _, resultado)
    resultado = a + b * c - d;

    // Expresion con parentesis: (a + b) * c
    // Esperado:
    //   (+, a, b, T4)
    //   (*, T4, c, T5)
    //   (:=, T5, _, resultado)
    resultado = (a + b) * c;

    // ==============================
    // 4. I/O: WRITE y READ
    // ==============================
    aqhe("Resultado: " & resultado);
    readline(a);

    // ==============================
    // 5. Condicional: IF / ELSE (JF, JMP, LABEL)
    // ==============================
    if (a >= 10) {
        aqhe("a es mayor o igual a 10");
    } else {
        aqhe("a es menor que 10");
    }

    // ==============================
    // 6. Bucle: WHILE (LABEL, JF, JMP)
    // ==============================
    contador = 0;
    while (contador < 5) {
        aqhe("Iteracion: " & contador);
        contador += 1;
    }

    // ==============================
    // 7. Try / Catch / Finally
    // ==============================
    // Esperado:
    //   (TRY_BEGIN, _, _, L#)     entrada primer catch
    //   (:=, 0, _, resultado)     cuerpo try
    //   (TRY_END, _, _, L#)       salta a finally
    //   (LABEL, _, _, L#)         entrada catch
    //   (:=, IOException, _, ex)  bind excepcion
    //   (WRITE, "Error de IO", _, _)
    //   (JMP, _, _, L#)           salta a finally
    //   (LABEL, _, _, L#)         etiqueta muerta
    //   (LABEL, _, _, L#)         finally
    //   (WRITE, "Fin del bloque", _, _)
    try {
        resultado = 0;
    } catch (IOException ex) {
        aqhe("Error de IO");
    } finally {
        aqhe("Fin del bloque");
    }

    // Dos catches sin finally:
    try {
        a = 1;
    } catch (IOException e1) {
        aqhe("IO");
    } catch (RuntimeException e2) {
        aqhe("Runtime");
    }
}
end
